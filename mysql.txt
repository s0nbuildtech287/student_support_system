CREATE DATABASE student_support


CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,

    name VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    password VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    avatar VARCHAR(255) DEFAULT '/static/images/avatar.jpg',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE subjects (
    id INT PRIMARY KEY,
    subject_name VARCHAR(255),
    category VARCHAR(100),
    career_group VARCHAR(100),
    level VARCHAR(50),
    prerequisite VARCHAR(255),
    study_hours INT,
    description TEXT,
    resource_type VARCHAR(50),
    resource_link TEXT,
    image VARCHAR(255)
);


CREATE TABLE study_plans (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    is_started TINYINT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)
);


CREATE TABLE plan_subjects (
    id INT AUTO_INCREMENT PRIMARY KEY,
    plan_id INT NOT NULL,
    subject_id INT NOT NULL,
    progress INT DEFAULT 0,
    status ENUM('not_started','studying','completed') DEFAULT 'not_started',

    FOREIGN KEY (plan_id) REFERENCES study_plans(id) ON DELETE CASCADE,
    FOREIGN KEY (subject_id) REFERENCES subjects(id),
    UNIQUE(plan_id, subject_id)
);
